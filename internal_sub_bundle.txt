const CAD_SCALE := 0.085

static func make_dual_mat(real_hex: int, bp_hex: int, rough := 0.5, metal := 0.25, tex: Texture2D = null, alpha := 1.0) -> StandardMaterial3D:
	var m := StandardMaterial3D.new()
	var bp_col := Color.hex((bp_hex << 8) | 0xFF)
	var real_col := Color.hex((real_hex << 8) | 0xFF)
	m.albedo_color = Color(bp_col.r, bp_col.g, bp_col.b, alpha)
	m.roughness = 0.88
	m.metallic = 0.04
	m.albedo_texture = tex
	if alpha < 0.99:
		m.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
		m.cull_mode = BaseMaterial3D.CULL_DISABLED
	m.set_meta("real_color", Color(real_col.r, real_col.g, real_col.b, alpha))
	m.set_meta("real_metal", metal)
	m.set_meta("real_rough", rough)
	m.set_meta("bp_color", Color(bp_col.r, bp_col.g, bp_col.b, alpha))
	m.set_meta("base_alpha", alpha)
	return m

static func add_part_mesh(part_node: Node3D, mesh: Mesh, material: StandardMaterial3D, pos: Vector3, rot := Vector3.ZERO, add_edges := true) -> MeshInstance3D:
	var mi := MeshInstance3D.new()
	mi.mesh = mesh
	mi.position = pos * CAD_SCALE
	mi.rotation = rot
	mi.material_override = material.duplicate()
	part_node.add_child(mi)

	if add_edges:
		var edge_mi := _create_edge_overlay(mesh)
		if edge_mi != null:
			mi.add_child(edge_mi)
	return mi

static func box(sz: Vector3) -> BoxMesh:
	var bm := BoxMesh.new()
	bm.size = sz * CAD_SCALE
	return bm

static func cyl(rt: float, rb: float, h: float, segs := 20) -> CylinderMesh:
	var cm := CylinderMesh.new()
	cm.top_radius = rt * CAD_SCALE
	cm.bottom_radius = rb * CAD_SCALE
	cm.height = h * CAD_SCALE
	cm.radial_segments = segs
	return cm

static func sphere(r: float, segs := 14) -> SphereMesh:
	var sm := SphereMesh.new()
	sm.radius = r * CAD_SCALE
	sm.height = r * 2.0 * CAD_SCALE
	sm.radial_segments = segs
	sm.rings = maxi(6, segs / 2)
	return sm

static func torus(inner_r: float, outer_r: float) -> TorusMesh:
	var tm := TorusMesh.new()
	tm.inner_radius = inner_r * CAD_SCALE
	tm.outer_radius = outer_r * CAD_SCALE
	return tm

static func _create_edge_overlay(mesh: Mesh) -> MeshInstance3D:
	var faces := mesh.get_faces()
	if faces.is_empty():
		return null
	var line_verts := PackedVector3Array()
	var n_tris := faces.size() / 3
	for t in range(n_tris):
		var a := faces[t * 3]
		var b := faces[t * 3 + 1]
		var c := faces[t * 3 + 2]
		line_verts.append(a); line_verts.append(b)
		line_verts.append(b); line_verts.append(c)
		line_verts.append(c); line_verts.append(a)
	var am := ArrayMesh.new()
	var arrays := []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = line_verts
	am.add_surface_from_arrays(Mesh.PRIMITIVE_LINES, arrays)
	var em := StandardMaterial3D.new()
	em.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	em.albedo_color = Color(0.06, 0.09, 0.16, 0.55)
	em.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	var edge_mi := MeshInstance3D.new()
	edge_mi.name = "CadEdgeLines"
	edge_mi.mesh = am
	edge_mi.material_override = em
	return edge_mi

func _ready() -> void:
	name = "InternalAssemblyRoot"
	floor_grid = ChassisShellBuilder.build_engineering_grid(self, -0.22)

	assembly_root = Node3D.new()
	assembly_root.name = "VP5231A_55Part_Assembly"

	guide_lines_mi = MeshInstance3D.new()
	guide_lines_mi.name = "AssemblyAxisGuideLines"
	var gm := StandardMaterial3D.new()
	gm.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	gm.albedo_color = Color(0.01, 0.52, 0.78, 0.65)
	gm.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	guide_lines_mi.material_override = gm
	add_child(guide_lines_mi)

	apply_explosion_factor(explode_factor)
	apply_render_style()

func apply_explosion_factor(factor: float) -> void:
	explode_factor = clampf(factor, 0.0, 1.0)
	for rec in part_records:
		var n: Node3D = rec["node"]
		n.position = rec["base_pos"] + rec["exp_offset"] * explode_factor
	_rebuild_guide_lines()

func set_guides_visible(vis: bool) -> void:
	show_guides = vis
	_rebuild_guide_lines()

func set_blueprint_mode(is_bp: bool) -> void:
	blueprint_mode = is_bp
	apply_render_style()

func set_selection(sel_id: int, hov_id: int, isolate: bool) -> void:
	selected_part_id = sel_id
	hovered_part_id = hov_id
	isolate_selected = isolate
	apply_render_style()

func _rebuild_guide_lines() -> void:
	if guide_lines_mi == null:
		return
	guide_lines_mi.visible = show_guides and explode_factor > 0.03
	if not guide_lines_mi.visible:
		return
	var verts := PackedVector3Array()
	for rec in part_records:
		var off: Vector3 = rec["exp_offset"]
		if off.length() > 0.01:
			var p0: Vector3 = rec["base_pos"]
			var p1: Vector3 = p0 + off * explode_factor
			var steps := 10
			for s in range(0, steps, 2):
				var t0 := float(s) / float(steps)
				var t1 := float(s + 1) / float(steps)
				verts.append(p0.lerp(p1, t0))
				verts.append(p0.lerp(p1, t1))
	if verts.is_empty():
		return
	var am := ArrayMesh.new()
	var arrays := []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = verts
	am.add_surface_from_arrays(Mesh.PRIMITIVE_LINES, arrays)
	guide_lines_mi.mesh = am

func apply_render_style() -> void:
	if floor_grid != null:
		floor_grid.visible = not blueprint_mode
	var active_id := selected_part_id if selected_part_id > 0 else hovered_part_id

	for rec in part_records:
		var pid: int = rec["id"]
		var n: Node3D = rec["node"]
		var is_sel := (pid == selected_part_id)
		var is_hov := (pid == hovered_part_id)
		var is_dim := (active_id > 0 and pid != active_id) or (isolate_selected and selected_part_id > 0 and pid != selected_part_id)

		_style_node_recursive(n, is_sel, is_hov, is_dim)

func _style_node_recursive(node: Node, is_sel: bool, is_hov: bool, is_dim: bool) -> void:
	for ch in node.get_children():
		if ch is MeshInstance3D:
			var mi := ch as MeshInstance3D
			if mi.name == "CadEdgeLines":
				mi.visible = blueprint_mode or is_sel or is_hov
				var em := mi.material_override as StandardMaterial3D
				if em != null:
					if is_sel:
						em.albedo_color = Color(0.01, 0.52, 0.78, 0.95)
					elif blueprint_mode:
						em.albedo_color = Color(0.06, 0.09, 0.16, 0.10 if is_dim else 0.75)
					else:
						em.albedo_color = Color(0.22, 0.74, 0.97, 0.10 if is_dim else 0.65)
			elif mi.material_override is StandardMaterial3D:
				var m := mi.material_override as StandardMaterial3D
				if m.has_meta("bp_color"):
					var base_c: Color = m.get_meta("bp_color") if blueprint_mode else m.get_meta("real_color")
					var alpha: float = 0.14 if is_dim else float(m.get_meta("base_alpha"))
					m.albedo_color = Color(base_c.r, base_c.g, base_c.b, alpha)
					m.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA if alpha < 0.99 else BaseMaterial3D.TRANSPARENCY_DISABLED
					m.metallic = 0.04 if blueprint_mode else float(m.get_meta("real_metal"))
					m.roughness = 0.88 if blueprint_mode else float(m.get_meta("real_rough"))
					if is_sel:
						m.emission_enabled = true
						m.emission = Color(0.01, 0.52, 0.78)
						m.emission_energy_multiplier = 0.45
					elif is_hov:
						m.emission_enabled = true
						m.emission = Color(0.22, 0.74, 0.97)
						m.emission_energy_multiplier = 0.32
					else:
						m.emission_enabled = false
		_style_node_recursive(ch, is_sel, is_hov, is_dim)

static func create_graticule_texture() -> ImageTexture:
	var w := 256
	var h := 208
	var img := Image.create(w, h, false, Image.FORMAT_RGBA8)
	var center := Vector2(w * 0.5, h * 0.5)
	for y in range(h):
		for x in range(w):
			var d := clampf(Vector2(x, y).distance_to(center) / 160.0, 0.0, 1.0)
			var col := Color(0.21, 0.66, 0.77).lerp(Color(0.07, 0.37, 0.46), d)
			img.set_pixel(x, y, col)

	var grid_col := Color(0.04, 0.14, 0.19, 0.78)
	for c in range(11):
		var gx := clampi(int(round(float(c) * w / 10.0)), 0, w - 1)
		for y in range(h):
			img.set_pixel(gx, y, grid_col)
	for r in range(9):
		var gy := clampi(int(round(float(r) * h / 8.0)), 0, h - 1)
		for x in range(w):
			img.set_pixel(x, gy, grid_col)

	# CH1 Sine Wave & CH2 Square Wave Phosphor Traces
	var ch1_col := Color(0.82, 0.98, 0.90, 1.0)
	var ch2_col := Color(0.43, 0.91, 0.72, 1.0)
	for x in range(w):
		var u := float(x) / float(w)
		var y1 := clampi(int(74.0 - sin(u * TAU * 3.0) * 26.0), 1, h - 2)
		var y2 := 138 if (int(floor(u * 6.0)) % 2 == 0) else 168
		for dy in range(-1, 2):
			img.set_pixel(x, y1 + dy, ch1_col)
			img.set_pixel(x, y2 + dy, ch2_col)

	return ImageTexture.create_from_image(img)

static func create_pcb_texture(base_col: Color, trace_col: Color) -> ImageTexture:
	var w := 256
	var h := 256
	var img := Image.create(w, h, false, Image.FORMAT_RGBA8)
	img.fill(base_col)

	for i in range(28):
		var y0 := 12 + i * 8
		var knee := 90 + (i % 7) * 10
		for x in range(10, w - 10):
			var y := y0 if x < knee else mini(h - 6, y0 + 10)
			img.set_pixel(x, y, trace_col)

	var pad_col := Color(0.85, 0.70, 0.24)
	for px in range(20, w - 16, 22):
		for py in range(20, h - 16, 22):
			for dx in range(-1, 2):
				for dy in range(-1, 2):
					img.set_pixel(px + dx, py + dy, pad_col)

	return ImageTexture.create_from_image(img)
