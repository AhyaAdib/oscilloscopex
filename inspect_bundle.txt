func recompute() -> void:
	var sweep_span: float = float(view_config["time_per_div"]) * 10.0
	var sweep_samples: int = int(view_config["sample_count"])
	var total_span := sweep_span * 3.0
	var total_samples := sweep_samples * 3

	var p1_bnc: String = str(analog_config.get("probe1_bnc", "CH1"))
	var p1_tip: String = str(analog_config.get("probe1_tip", "FG_CH1"))
	var p2_bnc: String = str(analog_config.get("probe2_bnc", "CH2"))
	var p2_tip: String = str(analog_config.get("probe2_tip", "FG_CH2"))

	var ch1_tip := "FLOATING"
	if p1_bnc == "CH1":
		ch1_tip = p1_tip
	elif p2_bnc == "CH1":
		ch1_tip = p2_tip

	var ch2_tip := "FLOATING"
	if p2_bnc == "CH2":
		ch2_tip = p2_tip
	elif p1_bnc == "CH2":
		ch2_tip = p1_tip

	var psu_on := bool(analog_config.get("psu_power_on", true))
	var psu_vcc := clampf(float(analog_config.get("psu_voltage_coarse", 12.0)) + float(analog_config.get("psu_voltage_fine", 0.0)), 0.0, 30.0) if psu_on else 0.0
	breadboard_config["psu_vcc"] = psu_vcc

	# CH 1 Analog Chain
	var raw1 := Breadboard3DBuilder.generate_breadboard_signal(breadboard_config, ch1_tip, total_samples, total_span, sim_time_offset) if ch1_tip.begins_with("BB_TP") else SignalGenerator.generate_buffer(_resolve_tip_signal(ch1_tip), total_samples, total_span, sim_time_offset)
	raw_time_buffer = raw1["times"]
	raw_voltage_buffer = raw1["voltages"]
	var dt: float = raw1["dt"]

	var analog1 := AnalogScopeEngine.process_chain(raw_time_buffer, raw_voltage_buffer, dt, analog_config, view_config)
	analog_nodes = analog1["nodes"]
	time_buffer = analog1["sweep_times"]
	voltage_buffer = analog1["sweep_volts"]
	is_triggered = analog1["triggered"]

	# CH 2 Analog Chain
	var raw2 := Breadboard3DBuilder.generate_breadboard_signal(breadboard_config, ch2_tip, total_samples, total_span, sim_time_offset) if ch2_tip.begins_with("BB_TP") else SignalGenerator.generate_buffer(_resolve_tip_signal(ch2_tip), total_samples, total_span, sim_time_offset)
	var cfg2 := analog_config.duplicate()
	cfg2["coupling"] = analog_config.get("ch2_coupling", AnalogScopeEngine.CouplingMode.DC)
	cfg2["vert_pos_div"] = analog_config.get("ch2_vert_pos_div", 0.0)
	cfg2["probe_attenuation"] = analog_config.get("probe2_attenuation", 1.0)
	var view2 := view_config.duplicate()
	view2["volts_per_div"] = view_config.get("ch2_volts_per_div", 1.0)
	var analog2 := AnalogScopeEngine.process_chain(raw2["times"], raw2["voltages"], dt, cfg2, view2)
	ch2_voltage_buffer = analog2["sweep_volts"]

	state_updated.emit()

func get_calibration_checklist() -> Array[Dictionary]:
	var p1_bnc: String = str(analog_config.get("probe1_bnc", "UNPLUGGED"))
	var p1_tip: String = str(analog_config.get("probe1_tip", "FLOATING"))
	var p2_bnc: String = str(analog_config.get("probe2_bnc", "UNPLUGGED"))
	var p2_tip: String = str(analog_config.get("probe2_tip", "FLOATING"))

	var step1_ok := (p1_bnc == "CH1" and p1_tip == "CAL_1VPP") or (p2_bnc == "CH1" and p2_tip == "CAL_1VPP")
	var step2_ok := (int(analog_config["coupling"]) == AnalogScopeEngine.CouplingMode.DC) and (absf(float(analog_config["vert_pos_div"])) <= 0.45)
	var trig_v := float(analog_config["trigger_level"])
	var step3_ok := is_triggered and (trig_v >= 0.15 and trig_v <= 0.85)
	var v_div := float(view_config["volts_per_div"])
	var t_div := float(view_config["time_per_div"])
	var step4_ok := (v_div <= 1.0) and (t_div >= 0.0002 and t_div <= 0.001)
	var comp := float(analog_config["probe_compensation"])
	var step5_ok := (float(analog_config["probe_attenuation"]) > 1.5) and (absf(comp - 1.0) <= 0.06)

	return [
		{"done": step1_ok, "text": "1. Plug Probe BNC into CH 1 & clip Tip onto CAL 1Vp-p"},
		{"done": step2_ok, "text": "2. Set CH 1 Coupling = DC & center V-Position (~0 div)"},
		{"done": step3_ok, "text": "3. Adjust Trigger LEVEL (0.2V–0.8V) to lock square wave"},
		{"done": step4_ok, "text": "4. Set VOLTS/DIV (0.5V/1V) & TIME/DIV (0.5ms) to frame wave"},
		{"done": step5_ok, "text": "5. Flip Probe to 10x & tune PROBE COMP for flat corners"}
	]

func get_node_stats(node_id: String) -> Dictionary:
	var buf: PackedFloat64Array = analog_nodes.get(node_id, PackedFloat64Array())
	if buf.is_empty():
		return {"v_min": 0.0, "v_max": 0.0, "v_pp": 0.0, "v_rms": 0.0, "v_inst": 0.0, "buffer": buf}
	var v_min := buf[0]
	var v_max := buf[0]
	var sum_sq := 0.0
	for v in buf:
		if v < v_min:
			v_min = v
		if v > v_max:
			v_max = v
		sum_sq += v * v
	return {
		"v_min": v_min,
		"v_max": v_max,
		"v_pp": v_max - v_min,
		"v_rms": sqrt(sum_sq / float(buf.size())),
		"v_inst": buf[buf.size() / 2],
		"buffer": buf
	}

func submit_mission_reading(mission: Dictionary, sub_vpp: float, sub_freq: float, sub_offset: float, time_left: float) -> Dictionary:
	var res := ChallengeManager.evaluate_submission(mission, sub_vpp, sub_freq, sub_offset, time_left)
	submission_attempts += 1
	if not res["passed"]:
		failed_attempts += 1
	var finished := bool(res["passed"]) or time_left <= 0.0
	if finished:
		active_challenge_running = false
		last_mission_report = "FINAL: %d/100 (%s) | Attempts: %d (Fails: %d) | True: %.2fVpp, %.0fHz, %.2fV" % [
			res["total_score"],
			"PASS" if res["passed"] else "TIME UP",
			submission_attempts,
			failed_attempts,
			res["true_vpp"],
			res["true_freq"],
			res["true_offset"]
		]
	else:
		last_mission_report = "Score: %d/100 — KEEP GOING! | Attempts: %d (Fails: %d) | Route:%s | Comp:%s | Trig:%s | Vpp:%s | Freq:%s" % [
			res["total_score"],
			submission_attempts,
			failed_attempts,
			"OK" if res["routing_ok"] else "NO",
			"OK" if res["comp_ok"] else "OFF",
			"OK" if res["trig_ok"] else "NO",
			"OK" if res["vpp_ok"] else "ERR",
			"OK" if res["freq_ok"] else "ERR"
		]
	var verdict := "[color=#4ade80][b]PASS — LAB PRACTICUM VERIFIED![/b][/color]" if res["passed"] else "[color=#f87171][b]NEEDS RE-CALIBRATION / RE-MEASUREMENT[/b][/color]"
	var lines: Array[String] = [
		"%s   |   [b]Score: %d / 100[/b] (Cal: %d/50, Reading: %d/50)" % [verdict, res["total_score"], res["cal_score"], res["read_score"]],
		"• Probe Routing: %s   |   DC Coupling: %s   |   Trigger Lock: %s   |   Probe Comp: %s" % [
			"OK" if res["routing_ok"] else "WRONG",
			"OK" if res["coupling_ok"] else "WRONG",
			"LOCKED" if res["trig_ok"] else "UNLOCKED",
			"OK" if res["comp_ok"] else "OFF"
		],
		"• Vpp: submitted [b]%.2f V[/b] vs true [b]%.2f V[/b] (err %.1f%%) — %s" % [sub_vpp, res["true_vpp"], res["vpp_err_pct"], "PASS" if res["vpp_ok"] else "FAIL"],
		"• Freq: submitted [b]%.0f Hz[/b] vs true [b]%.0f Hz[/b] (err %.1f%%) — %s" % [sub_freq, res["true_freq"], res["freq_err_pct"], "PASS" if res["freq_ok"] else "FAIL"],
		"• Offset: submitted [b]%.2f V[/b] vs true [b]%.2f V[/b] (err %.2f V) — %s" % [sub_offset, res["true_offset"], res["off_err_v"], "PASS" if res["off_ok"] else "FAIL"]
	]
	return {
		"result": res,
		"finished": finished,
		"report_lines": lines,
		"summary": "Score: %d/100 (%s)" % [res["total_score"], "PASS" if res["passed"] else "RETRY"]
	}

func _get_stage_rect(stage: Dictionary, canvas_size: Vector2) -> Rect2:
	var box_w := clampf(canvas_size.x * 0.135, 92.0, 145.0)
	var box_h := 66.0
	var center := Vector2(stage["pos"].x * canvas_size.x, stage["pos"].y * canvas_size.y)
	return Rect2(center - Vector2(box_w * 0.5, box_h * 0.5), Vector2(box_w, box_h))

func _on_diagram_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		var mb := event as InputEventMouseButton
		if mb.pressed and mb.button_index == MOUSE_BUTTON_LEFT:
			var sz := _diagram_canvas.size
			for stage in STAGES:
				if _get_stage_rect(stage, sz).has_point(mb.position):
					if has_node("/root/SimState"):
						SimState.select_circuit_node(stage["id"])
					break

func _draw_block_diagram() -> void:
	var sz := _diagram_canvas.size
	if sz.x < 40.0 or sz.y < 40.0:
		return
	var font := ThemeDB.fallback_font


  # Section Titles
	_diagram_canvas.draw_string(font, Vector2(18, 26), "NATIONAL VP-5231A — INTERNAL ANALOG SIGNAL & TIMEBASE ARCHITECTURE (Click Any Block to Probe)", HORIZONTAL_ALIGNMENT_LEFT, -1, 13, Color(0.75, 0.86, 0.92, 1.0))

	if has_node("/root/SimState"):
		power_on = bool(SimState.view_config.get("power_on", true))
		selected_id = SimState.selected_circuit_node

	# Draw Connecting Wires & Animated Signal Pulses
	for wire in WIRES:
		var r_from: Rect2 = stage_rects[wire[0]]
		var r_to: Rect2 = stage_rects[wire[1]]
		var branch: String = wire[2]
		var p1 := Vector2(r_from.end.x, r_from.get_center().y)
		var p2 := Vector2(r_to.position.x, r_to.get_center().y)
		if wire[0] == "N2_COUPLING" and wire[1] == "N7_TRIGGER":
			p1 = Vector2(r_from.get_center().x, r_from.end.y)
			p2 = Vector2(r_to.position.x, r_to.get_center().y)

		var base_col := Color(0.22, 0.75, 0.58, 0.75) if branch == "VERT" else Color(0.88, 0.65, 0.22, 0.75)
		if not power_on:
			base_col = Color(0.25, 0.28, 0.30, 0.45)

		if wire[0] == "N2_COUPLING" and wire[1] == "N7_TRIGGER":
			var elbow := Vector2(p1.x, p2.y)
			_diagram_canvas.draw_line(p1, elbow, base_col, 2.0)
			_diagram_canvas.draw_line(elbow, p2, base_col, 2.0)
			if power_on:
				var pulse_pos := p1.lerp(elbow, _flow_phase * 2.0) if _flow_phase < 0.5 else elbow.lerp(p2, (_flow_phase - 0.5) * 2.0)
				_diagram_canvas.draw_circle(pulse_pos, 4.0, Color(1.0, 0.85, 0.35, 1.0))
		else:
			_diagram_canvas.draw_line(p1, p2, base_col, 2.2)
			# Arrowhead
			_diagram_canvas.draw_colored_polygon(PackedVector2Array([
				p2,
				p2 + Vector2(-7, -4),
				p2 + Vector2(-7, 4)
			]), base_col)
			if power_on:
				var pulse_pos := p1.lerp(p2, _flow_phase)
				var dot_col := Color(0.45, 1.0, 0.82, 1.0) if branch == "VERT" else Color(1.0, 0.85, 0.35, 1.0)
				_diagram_canvas.draw_circle(pulse_pos, 4.0, dot_col)

	# Draw Stage Blocks with Live Vpp Readout
	for stage in STAGES:
		var id: String = stage["id"]
		var r: Rect2 = stage_rects[id]
		var is_sel := (id == selected_id)
		var is_vert: bool = (stage["branch"] == "VERT")

		var fill_col := Color(0.11, 0.16, 0.20, 1.0) if not is_sel else Color(0.14, 0.26, 0.30, 1.0)
		var border_col := (Color(0.30, 1.0, 0.72, 1.0) if is_vert else Color(1.0, 0.78, 0.28, 1.0)) if is_sel else Color(0.30, 0.40, 0.48, 0.9)

		_diagram_canvas.draw_string(font, r.position + Vector2(6, 33), stage["sub"], HORIZONTAL_ALIGNMENT_LEFT, r.size.x - 10, 10, Color(0.65, 0.75, 0.82, 1.0))

		if has_node("/root/SimState"):
			var stats := SimState.get_node_stats(id)
			var vpp_str := "Vpp: %.2f V" % stats["v_pp"]
			var val_col := Color(0.35, 1.0, 0.75, 1.0) if is_vert else Color(1.0, 0.82, 0.35, 1.0)
			_diagram_canvas.draw_string(font, r.position + Vector2(6, 54), vpp_str, HORIZONTAL_ALIGNMENT_LEFT, r.size.x - 10, 11, val_col)

	# Draw Schematic CRT Tube Symbol on Far Right
	var crt_rect := Rect2(Vector2(sz.x * 0.83, sz.y * 0.56), Vector2(clampf(sz.x * 0.13, 95.0, 140.0), 86.0))
	_diagram_canvas.draw_rect(crt_rect, Color(0.10, 0.34, 0.42, 1.0) if power_on else Color(0.08, 0.14, 0.16, 1.0), true)
	_diagram_canvas.draw_rect(crt_rect, Color(0.55, 0.85, 0.92, 0.9), false, 2.0)

func _refresh_inspector() -> void:
	if not has_node("/root/SimState") or _title_lbl == null:
		return
	var sel_id := SimState.selected_circuit_node
	var active_stage := STAGES[0]
	for s in STAGES:
		if s["id"] == sel_id:
			active_stage = s
			break

	_title_lbl.text = active_stage["title"]
	_eq_lbl.text = "Equation:  " + active_stage["equation"]
	_desc_lbl.text = active_stage["desc"]

	var coup_name: String = AnalogScopeEngine.COUPLING_NAMES[SimState.analog_config["coupling"]]
	var probe_x: int = int(SimState.analog_config["probe_attenuation"])
	var trig_lvl: float = SimState.analog_config["trigger_level"]
	var v_pos: float = SimState.analog_config["vert_pos_div"]
	_params_lbl.text = "Active State:  Coupling = %s  |  Probe = %dx  |  Trig = %.2f V  |  V-Pos = %.1f div" % [
		coup_name, probe_x, trig_lvl, v_pos
	]

	_metrics_lbl.text = "Vpp: %.3f V   |   Vrms: %.3f V   |   Min/Max: %.2fV / %.2fV" % [

	]
	if _diagram_canvas != null:
		_diagram_canvas.queue_redraw()
	if _probe_canvas != null:
		_probe_canvas.queue_redraw()

func _draw_node_probe_waveform() -> void:
	var sz := _probe_canvas.size
	if sz.x < 20.0 or sz.y < 20.0:
		return
	_probe_canvas.draw_rect(Rect2(Vector2.ZERO, sz), Color(0.03, 0.05, 0.07, 1.0), true)
	_probe_canvas.draw_rect(Rect2(Vector2.ZERO, sz), Color(0.22, 0.34, 0.42, 1.0), false, 1.5)

	# Zero-volt midline
	var mid_y := sz.y * 0.5
	_probe_canvas.draw_line(Vector2(0.0, mid_y), Vector2(sz.x, mid_y), Color(0.20, 0.30, 0.36, 0.9), 1.0)

	if not has_node("/root/SimState"):
		return
	var stats := SimState.get_node_stats(SimState.selected_circuit_node)
	var buf: PackedFloat64Array = stats["buffer"]
	var n := buf.size()
	if n < 2:
		return

	# Auto-scale vertical preview cleanly around 0V so small attenuated or large rail signals remain clear
	var peak_abs := maxf(maxf(absf(stats["v_min"]), absf(stats["v_max"])) * 1.25, 0.5)
	var px_per_v := (sz.y * 0.46) / peak_abs
	var step_x := sz.x / float(n - 1)

	var pts := PackedVector2Array()
	pts.resize(n)
	for i in range(n):
		pts[i] = Vector2(float(i) * step_x, mid_y - float(buf[i]) * px_per_v)

	var is_trig_branch := (SimState.selected_circuit_node == "N7_TRIGGER" or SimState.selected_circuit_node == "N8_SWEEP")
	var trace_col := Color(1.0, 0.80, 0.30, 1.0) if is_trig_branch else Color(0.30, 1.0, 0.72, 1.0)
	_probe_canvas.draw_polyline(pts, trace_col, 2.0, true)

	var font := ThemeDB.fallback_font
	_probe_canvas.draw_string(font, Vector2(8, 16), "LIVE NODE PROBE (±%.2f V full-scale)" % peak_abs, HORIZONTAL_ALIGNMENT_LEFT, -1, 10, Color(0.75, 0.85, 0.90, 0.9))

static func update_single_probe(probe_data: Dictionary, snaps: Dictionary, bnc_ch: String, tip_tgt: String, dragging_kind := "", drag_world := Vector3.ZERO) -> void:
	if probe_data.is_empty():
		return
	var idx: int = int(probe_data["idx"])
	var bnc_plug: Node3D = probe_data["bnc_plug"]
	var probe_tip: Node3D = probe_data["probe_tip"]

	# Position BNC Plug
	if dragging_kind == "bnc":
		bnc_plug.global_position = drag_world
		bnc_plug.rotation = Vector3(0.15, 0.0, 0.0)
	elif snaps.has(bnc_ch):
		var info: Dictionary = snaps[bnc_ch]
		var n: Node3D = info["node"]
		bnc_plug.global_position = n.global_position + info["offset"]
		bnc_plug.rotation = Vector3.ZERO
	else:
		# Rest neatly on ESD mat to left (Probe 1) and right (Probe 2) of breadboard
		var rest_x := -0.18 if idx == 1 else 0.16
		bnc_plug.global_position = Vector3(rest_x, 0.011, 0.22)
		bnc_plug.rotation = Vector3(0.0, 0.18 if idx == 1 else -0.18, 0.0)

	# Position Probe Tip
	if dragging_kind == "tip":
		probe_tip.global_position = drag_world
		probe_tip.rotation = Vector3(-0.20, 0.0, 0.0)
	elif snaps.has(tip_tgt):
		var info: Dictionary = snaps[tip_tgt]
		var n: Node3D = info["node"]
		var offset_v: Vector3 = info["offset"]
		# Cleanly seat hook pin onto CAL loop (slight vertical stack if both probes share CAL)
		if idx == 2 and tip_tgt == "CAL_1VPP":
			offset_v += Vector3(0.0, -0.0025, 0.001)
		probe_tip.global_position = n.global_position + offset_v
		probe_tip.rotation = Vector3(-0.08, -0.06 if tip_tgt == "CAL_1VPP" else -0.15, 0.0)
	else:
		# Rest neatly on ESD mat clear of breadboard without crossing cables
		var rest_tx := -0.14 if idx == 1 else 0.13
		probe_tip.global_position = Vector3(rest_tx, 0.011, 0.33)
		probe_tip.rotation = Vector3(0.0, 0.32 if idx == 1 else -0.32, 0.0)

	# Update 28-Segment 3D Coaxial Cable (scale applied AFTER look_at!)
	var p0 := bnc_plug.global_position + bnc_plug.global_transform.basis.z * 0.052
	var p1 := probe_tip.global_position + probe_tip.global_transform.basis.z * 0.096
	var segs: Array = probe_data["cable_segments"]
	var count := segs.size()

	var pts: Array[Vector3] = []
	for i in range(count + 1):
		var t := float(i) / float(count)
		var pt := p0.lerp(p1, t)
		pt.z += sin(t * PI) * (0.065 if idx == 1 else 0.090)
		pt.y = maxf(pt.y - sin(t * PI) * 0.045, 0.004)
		pts.append(pt)

	for i in range(count):
		var a := pts[i]
		var b := pts[i + 1]
		var seg: MeshInstance3D = segs[i]
		var diff := b - a
		var length := maxf(diff.length(), 1e-4)
		var mid := (a + b) * 0.5
		var up_dir := diff / length
		if absf(up_dir.dot(Vector3.UP)) > 0.99:
			seg.look_at_from_position(mid, b, Vector3.FORWARD)
		else:
			seg.look_at_from_position(mid, b, Vector3.UP)
		seg.rotate_object_local(Vector3.RIGHT, PI * 0.5)
		seg.scale = Vector3(1.0, length, 1.0)

static func _get_knob_dial_ticks(param_id: String, min_v: float, max_v: float, labels: Array) -> Array:
	if labels.size() > 0:
		return labels
	match param_id:
		"fg_frequency":
			return ["100", "1k", "2.5k", "4k", "5k"]
		"fg_amplitude":
			return ["0.2", "2", "4", "6", "8"]
		"fg_offset":
			return ["-3", "-1.5", "0", "+1.5", "+3"]
		"trigger_level", "vert_pos_div", "ch2_pos":
			return ["-4", "-2", "0", "+2", "+4"]
		"horiz_pos_div":
			return ["-5", "-2.5", "0", "+2.5", "+5"]
		"probe_compensation":
			return ["0.65", "1.0", "1.35"]
		_:
			return ["0", "5", "10"]

	labels: Array,
	interactables: Array[Dictionary]
) -> Node3D:
	var holder := Node3D.new()
	holder.name = "Knob_" + param_id
	holder.position = pos
	parent.add_child(holder)

	var is_fg := param_id.begins_with("fg_")
	var is_big_dial := param_id in ["time_per_div", "volts_per_div", "ch2_volts"]
	var tick_col := Color(0.92, 0.94, 0.98) if is_fg else COL_SILKSCREEN
	var title_y := radius + (0.0118 if is_big_dial else 0.0096)
	_add_label_3d(holder, title, Vector3(0.0, title_y, 0.0012), 14, tick_col)

	var skirt_mat := _mat(Color(0.86, 0.84, 0.80, 1.0), 0.4, 0.2)
	_add_cyl(holder, "Skirt", radius * 1.12, 0.003, Vector3(0, 0, 0.0015), Vector3(PI * 0.5, 0, 0), skirt_mat)

	# Circular engraved dial ring around TIME/DIV and VOLTS/DIV (matching real National VP-5231A)
	if is_big_dial:
		var ring_mi := MeshInstance3D.new()
		var torus := TorusMesh.new()
		torus.inner_radius = radius + 0.0068
		torus.outer_radius = radius + 0.0076
		torus.rings = 32
		torus.ring_segments = 8
		ring_mi.mesh = torus
		ring_mi.rotation.x = PI * 0.5
		ring_mi.position = Vector3(0.0, 0.0, 0.0008)
		ring_mi.material_override = _mat(COL_SILKSCREEN, 0.6, 0.0)
		holder.add_child(ring_mi)
		if param_id == "time_per_div":
			var u_ms := _add_label_3d(holder, "ms", Vector3(-radius - 0.0078, radius + 0.0055, 0.0012), 11, COL_SILKSCREEN)
			u_ms.pixel_size = 0.00018
			var u_us := _add_label_3d(holder, "µs", Vector3(radius + 0.0078, radius + 0.0055, 0.0012), 11, COL_SILKSCREEN)
			u_us.pixel_size = 0.00018
		else:
			var u_v := _add_label_3d(holder, "V", Vector3(-radius - 0.0075, radius + 0.0052, 0.0012), 11, COL_SILKSCREEN)
			u_v.pixel_size = 0.00018

	# Radial scale numbers & tick marks around the knob dial (135 deg CCW -> -135 deg CW)
	var dial_ticks := _get_knob_dial_ticks(param_id, min_v, max_v, labels)
	var tick_count := dial_ticks.size()
	var tick_r := radius + 0.0024
	var lbl_r := radius + (0.0049 if is_big_dial else 0.0058)
	for ti in range(tick_count):
		var frac := float(ti) / float(maxi(tick_count - 1, 1))
		var ang := lerpf(deg_to_rad(135.0), deg_to_rad(-135.0), frac)
		var dir := Vector2(-sin(ang), cos(ang))
		var t_box := _add_box(holder, "Tick_" + str(ti), Vector3(0.0006, 0.0016, 0.0008), Vector3(dir.x * tick_r, dir.y * tick_r, 0.0012), _mat(tick_col, 0.5, 0.1))
		t_box.rotation.z = ang
		var l_node := _add_label_3d(holder, str(dial_ticks[ti]), Vector3(dir.x * lbl_r, dir.y * lbl_r, 0.0015), 10, tick_col)
		l_node.pixel_size = 0.00016

	var rotor := Node3D.new()
	rotor.name = "Rotor"
	rotor.position = Vector3(0, 0, 0.003)
	holder.add_child(rotor)

	_add_cyl(rotor, "Body", radius, 0.012, Vector3(0, 0, 0.006), Vector3(PI * 0.5, 0, 0), _mat(COL_KNOB_DARK, 0.55, 0.1))
	_add_cyl(rotor, "Cap", radius * 0.62, 0.002, Vector3(0, 0, 0.0125), Vector3(PI * 0.5, 0, 0), _mat(cap_color, 0.45, 0.15))
	_add_box(rotor, "Pointer", Vector3(0.0018, radius * 0.88, 0.0020), Vector3(0.0, radius * 0.44, 0.0132), _mat(Color(0.96, 0.95, 0.92, 1.0), 0.3, 0.0))

	var norm := clampf((init_v - min_v) / maxf(max_v - min_v, 1e-6), 0.0, 1.0)
	rotor.rotation.z = lerpf(deg_to_rad(135.0), deg_to_rad(-135.0), norm)

	interactables.append({
		"type": "knob",
		"param_id": param_id,
		"title": title,
		"rotor": rotor,
		"focus_node": rotor,
		"hit_radius": maxf(radius * 1.65, 0.014),
		"ring_radius": radius * 1.35,
		"min": min_v,
		"max": max_v,
		"step": step_v,
		"value": init_v,
		"labels": labels
	})
	return rotor

static func _add_slide_switch_3d(
	parent: Node3D,
	param_id: String,
	title: String,
	pos: Vector3,
	options: Array,
	init_idx: int,
	vertical: bool,
	interactables: Array[Dictionary]
) -> Node3D:
	var holder := Node3D.new()
	holder.name = "Switch_" + param_id
	holder.position = pos
	parent.add_child(holder)

	_add_label_3d(holder, title, Vector3(0.0, 0.0135 if vertical else 0.0095, 0.001), 14, COL_SILKSCREEN)

	var slot_size := Vector3(0.006, 0.018, 0.002) if vertical else Vector3(0.022, 0.006, 0.002)
	_add_box(holder, "Slot", slot_size, Vector3(0, 0, 0.001), _mat(Color(0.14, 0.14, 0.14, 1.0), 0.8, 0.1))

	var nub := _add_box(holder, "Actuator", Vector3(0.0055, 0.0055, 0.0075), Vector3(0, 0, 0.004), _mat(Color(0.85, 0.22, 0.18, 1.0) if param_id == "probe_attenuation_switch" else Color(0.28, 0.27, 0.26, 1.0), 0.4, 0.3))

	var meta := {
		"type": "switch",
		"param_id": param_id,
		"title": title,
		"actuator": nub,
		"focus_node": nub,
		"hit_radius": 0.014,
		"ring_radius": 0.012,
		"options": options,
		"index": init_idx,
		"vertical": vertical
	}
	update_switch_visual(meta)
	interactables.append(meta)
	return nub

static func update_switch_visual(meta: Dictionary) -> void:
	var nub: Node3D = meta["actuator"]
	var count: int = maxi(meta["options"].size(), 2)
	var idx: int = clampi(int(meta["index"]), 0, count - 1)
	var frac := float(idx) / float(count - 1)
	var offset := lerpf(-0.006, 0.006, frac)
	if meta["vertical"]:
		nub.position.y = -offset
	else:
		nub.position.x = offset

static func _add_push_button(parent: Node3D, param_id: String, title: String, pos: Vector3, interactables: Array[Dictionary]) -> MeshInstance3D:
	var holder := Node3D.new()
	holder.position = pos
	parent.add_child(holder)

	var cap := _add_box(holder, "ButtonCap", Vector3(0.010, 0.010, 0.008), Vector3(0, 0, 0.003), _mat(Color(0.18, 0.17, 0.17, 1.0), 0.45, 0.2))

	interactables.append({
		"type": "button",
		"param_id": param_id,
		"title": title,
		"cap": cap,
		"focus_node": cap,
		"hit_radius": 0.013,
		"ring_radius": 0.010
	})
	return cap

static func _add_bnc_jack(parent: Node3D, jack_name: String, pos: Vector3, scale_f := 1.0) -> Node3D:
	var holder := Node3D.new()
	holder.name = jack_name
	holder.position = pos
	parent.add_child(holder)

	var metal_mat := _mat(COL_METAL_BNC, 0.25, 0.85)
	var insulator_mat := _mat(Color(0.92, 0.92, 0.90, 1.0), 0.35, 0.0)
	_add_cyl(holder, "Flange", 0.0075 * scale_f, 0.003 * scale_f, Vector3(0, 0, 0.0015), Vector3(PI * 0.5, 0, 0), metal_mat)
	_add_cyl(holder, "Barrel", 0.0052 * scale_f, 0.012 * scale_f, Vector3(0, 0, 0.007 * scale_f), Vector3(PI * 0.5, 0, 0), metal_mat)
	_add_cyl(holder, "Dielectric", 0.0032 * scale_f, 0.0125 * scale_f, Vector3(0, 0, 0.0072 * scale_f), Vector3(PI * 0.5, 0, 0), insulator_mat)
	return holder

static func _add_box(parent: Node3D, node_name: String, sz: Vector3, pos: Vector3, mat: Material) -> MeshInstance3D:
	var mi := MeshInstance3D.new()
	mi.name = node_name
	var bm := BoxMesh.new()
	bm.size = sz
	mi.mesh = bm
	mi.material_override = mat
	mi.position = pos
	parent.add_child(mi)
	return mi

static func _add_cyl(parent: Node3D, node_name: String, r: float, h: float, pos: Vector3, rot: Vector3, mat: Material) -> MeshInstance3D:
	var mi := MeshInstance3D.new()
	mi.name = node_name
	var cm := CylinderMesh.new()
	cm.top_radius = r
	cm.bottom_radius = r
	cm.height = h
	cm.radial_segments = 24
	mi.mesh = cm
	mi.material_override = mat
	mi.position = pos
	mi.rotation = rot
	parent.add_child(mi)
	return mi

static func _add_label_3d(parent: Node3D, text: String, pos: Vector3, px_size: int, col: Color) -> Label3D:
	var l := Label3D.new()
	l.text = text
	l.pixel_size = 0.00024
	l.font_size = px_size
	l.outline_size = 0
	l.modulate = col
	l.no_depth_test = false
	l.shaded = false
	l.position = pos
	parent.add_child(l)
	return l

static func _mat(col: Color, rough: float, metal: float) -> StandardMaterial3D:
	var m := StandardMaterial3D.new()
	m.albedo_color = col
	m.roughness = rough
	m.metallic = metal
	return m
